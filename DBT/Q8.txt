8. Exception handling in Stored Procedure Exception handling with various handler actions.

mysql> CREATE PROCEDURE ContinueHandlerDemo()
    -> BEGIN
    ->     DECLARE CONTINUE HANDLER FOR SQLEXCEPTION
    ->     SELECT 'Error Occurred, Continuing Execution' AS Message;
    ->
    ->     INSERT INTO Employee VALUES
    ->     (101,'Priya',55000);
    ->
    ->     SELECT 'Procedure Completed' AS Status;
    -> END //
