2)
delimiter //

create procedure check_salary(in eid int)
begin
declare sal int;
select salary into sal from employee where empid=eid;

if sal>=50000 then
select 'high salary' as result;
else
select 'low salary' as result;
end if;
end//

delimiter ;


call check_salary(4);

+-------------+
| result      |
+-------------+
| high salary |
+-------------+

call check_salary(1);

+------------+
| result     |
+------------+
| low salary |
+------------+

delimiter //

create procedure grade_salary(in eid int)
begin
declare sal int;
select salary into sal from employee where empid=eid;

if sal>=50000 then
select 'grade a' as grade;
elseif sal>=35000 then
select 'grade b' as grade;
else
select 'grade c' as grade;
end if;
end//

delimiter ;


call grade_salary(6);

+---------+
| grade   |
+---------+
| grade a |
+---------+
